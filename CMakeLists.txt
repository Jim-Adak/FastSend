cmake_minimum_required(VERSION 3.21)
project(FastSend VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Network Widgets)
add_library(fastsend_core src/protocol.cpp src/transfer.cpp src/discovery.cpp)
target_include_directories(fastsend_core PUBLIC src)
target_link_libraries(fastsend_core PUBLIC Qt6::Core Qt6::Network)
if(MSVC)
  add_compile_options(/utf-8 /W4)
  target_compile_options(fastsend_core PRIVATE /utf-8 /W4)
endif()
qt_add_executable(FastSend src/main.cpp src/mainwindow.cpp src/mainwindow.h)
target_link_libraries(FastSend PRIVATE fastsend_core Qt6::Widgets)
set_target_properties(FastSend PROPERTIES WIN32_EXECUTABLE TRUE)
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(fastsend_tests tests/transfer_test.cpp)
  target_link_libraries(fastsend_tests PRIVATE fastsend_core Qt6::Test)
  add_test(NAME transfer_tests COMMAND fastsend_tests)
  set_tests_properties(transfer_tests PROPERTIES TIMEOUT 60)
endif()
install(TARGETS FastSend RUNTIME DESTINATION .)
